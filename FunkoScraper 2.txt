                System.out.println("Aviso: No se pudo encontrar el titulo. Es posible que Amazon haya saltado con un bloqueo o CAPTCHA");
            }
        } catch (IOException e) {
            System.out.println("Error de red al conectar con Amazon, posible bloqueo o sin internet: " +e.getMessage());
        }

        return Funkoslist;

    //Método para pasar precio en texto a número decimal
    private double convertirPrecioDouble(String textoPrecio) {
        try {
            /* Limpiamos símbolos y espacios
            y cambiamos comas por puntos*/
            String limpio = textoPrecio.replace("€", "").replace("$", "").trim().replace(",", ".");
            return  Double.parseDouble(limpio);
        } catch (NumberFormatException e) {
            return 0.0; //Devolvemos si falla
        }
    }
