package com.funko;

//Importamos las libreías

import kong.unirest.HttpResponse;
import kong.unirest.JsonNode;
import kong.unirest.Unirest;

public class Main {

    //Token del Botfather de Telegram
    private static final String BOT_TOKEN = "";

    //Mi Chat ID de Telegram
    private static final String CHAT_ID = "";

    public static void main(String[] args) {
        //mensaje que debería llegar a Telegram
        String message = "¡Hola! tu bot de seguimiento de ofertas funkos en Java ya está conecctado correctamente";

        // %s se sustituirá por nuestro token
        String url = String.format("https://api.telegram.org/bot%s/sendMessage",BOT_TOKEN);

        //Petición Post para envíar datos
        HttpResponse<JsonNode> response = Unirest.post(url)
                .field("chat_id", CHAT_ID)//usuario dirigido
                .field("text", message) //mensaje a enviar
                .asJson(); //resuesta de Telegram a formato JSON

        if (response.getStatus() == 200) { //200 es OK en el protocolo HTTP
            System.out.println("¡Mensaje enviado con éxito a Telegram!");
        } else {
            System.out.println("Error al enviar el mensaje: " + response.getBody().toPrettyString());
        }
    }
}
